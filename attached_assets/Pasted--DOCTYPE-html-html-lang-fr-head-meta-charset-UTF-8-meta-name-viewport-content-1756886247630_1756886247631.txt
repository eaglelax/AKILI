        // Event listeners
        function initializeEventListeners() {
            // Mobile menu
            const mobileMenuBtn = document.getElementById('mobileMenuBtn');
            const sidebar = document.getElementById('sidebar');
            const mobileOverlay = document.getElementById('mobileOverlay');

            mobileMenuBtn.addEventListener('click', () => {
                sidebar.classList.toggle('hidden');
                mobileOverlay.classList.toggle('hidden');
            });

            mobileOverlay.addEventListener('click', () => {
                sidebar.classList.add('hidden');
                mobileOverlay.classList.add('hidden');
            });

            // Filter buttons
            document.querySelectorAll('.filter-button').forEach(button => {
                button.addEventListener('click', (e) => {
                    document.querySelectorAll('.filter-button').forEach(b => b.classList.remove('active'));
                    e.target.classList.add('active');
                    currentFilter = e.target.dataset.filter;
                    applyFilters();
                });
            });

            // Search input
            document.getElementById('searchInput').addEventListener('input', (e) => {
                applyFilters();
            });

            // Sort select
            document.getElementById('sortSelect').addEventListener('change', (e) => {
                currentSort = e.target.value;
                applySorting();
            });

            // Export button
            document.getElementById('exportBtn').addEventListener('click', exportReport);

            // Add member button
            document.getElementById('addMemberBtn').addEventListener('click', () => {
                showNotification('Fonctionnalité à venir : Ajout de nouveau membre', 'info');
            });

            // Modal close
            document.querySelector('.close').addEventListener('click', closeModal);
            window.addEventListener('click', (e) => {
                const modal = document.getElementById('memberModal');
                if (e.target === modal) {
                    closeModal();
                }
            });

        // Helper functions
        function getWorkloadColor(workload) {
            if (workload < 60) return 'workload-low';
            if (workload < 80) return 'workload-medium';
            if (workload < 95) return 'workload-high';
            return 'workload-critical';
        }

        function generateStars(rating) {
            const fullStars = Math.floor(rating);
            const hasHalfStar = rating % 1 >= 0.5;
            let stars = '';

            for (let i = 0; i < fullStars; i++) {
                stars += '<svg class="star filled" fill="currentColor" viewBox="0 0 24 24"><path d="M12 2l3.09 6.26L22 9.27l-5 4.87 1.18 6.88L12 17.77l-6.18 3.25L7 14.14 2 9.27l6.91-1.01L12 2z"/></svg>';
            }

            if (hasHalfStar) {
                stars += '<svg class="star filled" fill="currentColor" viewBox="0 0 24 24"><path d="M12 2l3.09 6.26L22 9.27l-5 4.87 1.18 6.88L12 17.77l-6.18 3.25L7 14.14 2 9.27l6.91-1.01L12 2z" fill-opacity="0.5"/></svg>';
            }

            const emptyStars = 5 - Math.ceil(rating);
            for (let i = 0; i < emptyStars; i++) {
                stars += '<svg class="star" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M12 2l3.09 6.26L22 9.27l-5 4.87 1.18 6.88L12 17.77l-6.18 3.25L7 14.14 2 9.27l6.91-1.01L12 2z"/></svg>';
            }

            return stars;
        }

        // Filter and sort functions
        function applyFilters() {
            const searchTerm = document.getElementById('searchInput').value.toLowerCase();
            
            filteredMembers = teamMembers.filter(member => {
                // Search filter
                const matchesSearch = member.name.toLowerCase().includes(searchTerm) ||
                                    member.position.toLowerCase().includes(searchTerm) ||
                                    member.specialties.some(s => s.toLowerCase().includes(searchTerm));

                // Category filter
                let matchesFilter = true;
                switch (currentFilter) {
                    case 'available':
                        matchesFilter = member.status === 'available';
                        break;
                    case 'busy':
                        matchesFilter = member.status === 'busy';
                        break;
                    case 'senior':
                        matchesFilter = member.category === 'senior' || member.category === 'direction';
                        break;
                    case 'junior':
                        matchesFilter = member.category === 'junior';
                        break;
                    case 'design':
                        matchesFilter = member.skills.some(skill => 
                            ['Photoshop', 'Design', 'UI/UX', 'Motion Design', 'Graphisme'].includes(skill.name)
                        );
                        break;
                    case 'strategy':
                        matchesFilter = member.skills.some(skill => 
                            ['Stratégie', 'Management', 'Rédaction', 'Conceptualisation'].includes(skill.name)
                        );
                        break;
                }

                return matchesSearch && matchesFilter;
            });

            applySorting();
        }

        function applySorting() {
            filteredMembers.sort((a, b) => {
                switch (currentSort) {
                    case 'name':
                        return a.name.localeCompare(b.name);
                    case 'workload':
                        return b.workload - a.workload;
                    case 'performance':
                        return b.performance - a.performance;
                    case 'rate':
                        return b.rate - a.rate;
                    default:
                        return 0;
                }
            });

            renderTeamMembers();
            updateMetrics();
        }

        // Update metrics
        function updateMetrics() {
            const available = filteredMembers.filter(m => m.status === 'available').length;
            const busy = filteredMembers.filter(m => m.status === 'busy').length;
            const avgWorkload = Math.round(filteredMembers.reduce((sum, m) => sum + m.workload, 0) / filteredMembers.length);

            document.getElementById('totalMembers').textContent = filteredMembers.length;
            document.getElementById('availableMembers').textContent = available;
            document.getElementById('busyMembers').textContent = busy;
            document.getElementById('avgWorkload').textContent = avgWorkload + '%';
        }

                'away': 'bg-gray-100 text-gray-800',
                'leave': 'bg-red-100 text-red-800'
            };

        }

        // Export functionality
        function exportReport() {
            const reportData = {
                date: new Date().toLocaleDateString('fr-FR'),
                totalMembers: teamMembers.length,
                available: teamMembers.filter(m => m.status === 'available').length,
                busy: teamMembers.filter(m => m.status === 'busy').length,
                avgWorkload: Math.round(teamMembers.reduce((sum, m) => sum + m.workload, 0) / teamMembers.length),
                members: teamMembers.map(m => ({
                    name: m.name,
                    position: m.position,
                    rate: m.rate,
                    workload: m.workload,
                    performance: m.performance,
                    status: getStatusLabel(m.status),
                    projects: m.projects.join(', ')
                }))
            };

            const csvContent = generateCSV(reportData);
            downloadCSV(csvContent, `rapport_equipe_jofe_${new Date().toISOString().split('T')[0]}.csv`);
            showNotification('Rapport exporté avec succès', 'success');
        }

        function generateCSV(data) {
            const headers = ['Nom', 'Poste', 'Taux (FCFA/h)', 'Charge (%)', 'Performance', 'Statut', 'Projets'];
            const rows = data.members.map(m => [
                m.name, m.position, m.rate, m.workload, m.performance, m.status, m.projects
            ]);
            
            return [headers, ...rows].map(row => row.join(',')).join('\n');
        }

        function downloadCSV(content, filename) {
            const blob = new Blob([content], { type: 'text/csv;charset=utf-8;' });
            const link = document.createElement('a');
            const url = URL.createObjectURL(blob);
            link.setAttribute('href', url);
            link.setAttribute('download', filename);
            link.style.visibility = 'hidden';
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);
        }

        // Notification system
        function showNotification(message, type = 'info') {
            const toast = document.getElementById('notificationToast');
            const messageEl = document.getElementById('notificationMessage');
            
            messageEl.textContent = message;
            toast.className = `notification-toast ${type === 'success' ? '' : type === 'info' ? 'bg-blue-500' : 'bg-orange-500'}`;
            toast.classList.add('show');
            
            setTimeout(() => {
                toast.classList.remove('show');
            }, 3000);
        }
