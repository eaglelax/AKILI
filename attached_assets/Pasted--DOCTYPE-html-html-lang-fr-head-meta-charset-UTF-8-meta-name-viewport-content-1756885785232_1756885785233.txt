                    const dependency = projectData.edges.find(

                });
            });

        function applyFilters() {
            const projectFilter = document.getElementById('projectFilter').value;
            const statusFilter = document.getElementById('statusFilter').value;
            
            let filteredNodes = projectData.nodes;
            let filteredEdges = projectData.edges;
            
            // Apply project filter
            if (projectFilter !== 'all') {
                const projectMap = {
                    'moov': 'MOOV AFRICA',
                    'boa': 'BANK OF AFRICA',
                    'sunu': 'SUNU BURKINA',
                    'roxgold': 'ROXGOLD'
                };
                
                filteredNodes = filteredNodes.filter(node => 
                    node.project === projectMap[projectFilter]
                );
                
                const nodeIds = filteredNodes.map(node => node.id);
                filteredEdges = filteredEdges.filter(edge => 
                    nodeIds.includes(edge.from) && nodeIds.includes(edge.to)
                );
            }
            
            // Apply status filter
            if (statusFilter !== 'all') {
                if (statusFilter === 'critical') {
                    filteredNodes = filteredNodes.filter(node => node.critical);
                    filteredEdges = filteredEdges.filter(edge => edge.critical);
                } else if (statusFilter === 'conflict') {
                    // Show nodes with conflicts (simulated)
                    filteredNodes = filteredNodes.filter(node => 
                        node.id === 2 || node.id === 4 || node.id === 6
                    );
                } else {
                    filteredNodes = filteredNodes.filter(node => node.status === statusFilter);
                }
                
                const nodeIds = filteredNodes.map(node => node.id);
                filteredEdges = filteredEdges.filter(edge => 
                    nodeIds.includes(edge.from) && nodeIds.includes(edge.to)
                );
            }
            
            // Update network
            nodes.clear();
            edges.clear();
            
            nodes.add(filteredNodes.map(node => ({
                id: node.id,
                label: node.label,
                group: node.group,
                color: getNodeColor(node),
                font: { color: '#162C54', size: 12, face: 'Open Sans' },
                borderWidth: node.critical ? 3 : 1,
                shadow: true,
                title: getNodeTooltip(node)
            })));
            
            edges.add(filteredEdges.map((edge, index) => ({
                id: index,
                from: edge.from,
                to: edge.to,
                arrows: 'to',
                color: edge.critical ? '#F68C1F' : '#3475BB',
                width: edge.critical ? 3 : 1,
                dashes: edge.type === 'resource' ? [5, 5] : false,
                title: getEdgeTooltip(edge)
            })));
            
            showNotification('Filtres appliqués', 'success');
        }

        // Initialize drag and drop functionality
        function initializeDragDrop() {
            const taskCards = document.querySelectorAll('.task-card');
            const dropZones = document.querySelectorAll('.drop-zone');
            
            taskCards.forEach(card => {
                card.draggable = true;
                
                card.addEventListener('dragstart', function(e) {
                    this.classList.add('dragging');
                    e.dataTransfer.setData('text/plain', this.dataset.taskId);
                });
                
                card.addEventListener('dragend', function() {
                    this.classList.remove('dragging');
                });
            });
            
            dropZones.forEach(zone => {
                zone.addEventListener('dragover', function(e) {
                    e.preventDefault();
                    this.classList.add('drag-over');
                });
                
                zone.addEventListener('dragleave', function() {
                    this.classList.remove('drag-over');
                });
                
                zone.addEventListener('drop', function(e) {
                    e.preventDefault();
                    this.classList.remove('drag-over');
                    
                    const taskId = e.dataTransfer.getData('text/plain');
                    showNotification(`Tâche ${taskId} déplacée`, 'success');
                });
            });
        }

        // Initialize on load
        window.addEventListener('load', function() {
            initializeDragDrop();
        });
