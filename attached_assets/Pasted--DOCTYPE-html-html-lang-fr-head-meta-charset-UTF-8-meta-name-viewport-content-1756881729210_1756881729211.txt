        function toggleTeamMember(memberElement) {
            const memberId = memberElement.dataset.member;

                // Remove member
                memberElement.classList.remove('selected');

                    name: teamMembers[memberId].name,
                    role: teamMembers[memberId].role,
                    rate: memberRate

            calculateBudget();

        // Budget Calculations
        function calculateBudget() {
            const totalBudget = parseInt(document.getElementById('totalBudget').value) || 0;
            const estimatedHours = parseInt(document.getElementById('estimatedHours').value) || 0;
            
            let teamCost = 0;
            if (assignedTeamMembers.length > 0 && estimatedHours > 0) {
                const avgRate = assignedTeamMembers.reduce((sum, m) => sum + m.rate, 0) / assignedTeamMembers.length;
                teamCost = Math.round(avgRate * estimatedHours);
            }
            
            const profit = totalBudget - teamCost;
            const marginRate = totalBudget > 0 ? Math.round((profit / totalBudget) * 100) : 0;
            
            document.getElementById('teamEstimatedCost').textContent = teamCost.toLocaleString() + ' FCFA';
            document.getElementById('profitMargin').textContent = profit.toLocaleString() + ' FCFA';
            document.getElementById('marginRate').textContent = marginRate + '%';
            document.getElementById('projectTotal').textContent = totalBudget.toLocaleString() + ' FCFA';
        }

            if (draft) {
                try {
                    const data = JSON.parse(draft);
                    
                    // Restore form values
                    Object.keys(data.formValues).forEach(key => {
                        const element = document.getElementById(key);
                        if (element) {
                            element.value = data.formValues[key] || '';
                        }
                    });
                    
                    // Restore team members
                    if (data.assignedTeamMembers) {
                        data.assignedTeamMembers.forEach(member => {
                            const memberElement = document.querySelector(`[data-member="${member.id}"]`);
                            if (memberElement) {
                                memberElement.classList.add('selected');
                            }
                        });
                        assignedTeamMembers = data.assignedTeamMembers;
                        updateAssignedTeamDisplay();
                    }
                    
                    // Restore current step
                    if (data.currentStep) {
                        currentStep = data.currentStep;
                        updateStepDisplay();
                    }
                    
                    calculateBudget();
                    updateProjectSummary();
                    
                } catch (e) {
                    console.error('Error loading draft:', e);
                }
            }
        }

        // Submit Project
        function submitProject(e) {
            e.preventDefault();
            
            if (!validateCurrentStep()) {
                return;
            }
            
            // Collect all form data
            const projectData = {
                name: document.getElementById('projectName').value,
                client: document.getElementById('projectClient').value,
                type: document.getElementById('projectType').value,
                priority: document.getElementById('projectPriority').value,
                description: document.getElementById('projectDescription').value,
                objectives: document.getElementById('projectObjectives').value,
                team: assignedTeamMembers,
                budget: parseInt(document.getElementById('totalBudget').value),
                estimatedHours: parseInt(document.getElementById('estimatedHours').value),
                startDate: document.getElementById('startDate').value,
                endDate: document.getElementById('endDate').value,
                createdAt: new Date().toISOString(),
                createdBy: document.getElementById('currentUserName').textContent
            };
            
            // Save to localStorage (in real app, this would be sent to server)
            const existingProjects = JSON.parse(localStorage.getItem('jofePlus_projects') || '[]');
            projectData.id = 'proj_' + Date.now();
            existingProjects.push(projectData);
            localStorage.setItem('jofePlus_projects', JSON.stringify(existingProjects));
            
            // Clear draft
            localStorage.removeItem('jofePlus_projectDraft');
            
            showNotification('Projet créé avec succès !', 'success');
            
            // Redirect or reset form
            setTimeout(() => {
                if (confirm('Voulez-vous créer un autre projet ?')) {
                    location.reload();
                } else {
                    // Redirect to projects list
                    window.history.back();
                }
            }, 2000);
        }

        // Notifications
        function showNotification(message, type = 'info') {
            const notification = document.createElement('div');
            notification.className = `notification ${type}`;
            notification.textContent = message;
            
            document.body.appendChild(notification);
            
            setTimeout(() => {
                notification.remove();
            }, 4000);
        }

        // Auto-update dates
        document.getElementById('startDate').addEventListener('change', function() {
            const startDate = new Date(this.value);
            const estimatedHours = parseInt(document.getElementById('estimatedHours').value) || 40;
            const workingDaysNeeded = Math.ceil(estimatedHours / 8); // 8 hours per day
            
            if (startDate) {
                const endDate = new Date(startDate);
                endDate.setDate(endDate.getDate() + workingDaysNeeded + 2); // Add buffer
                document.getElementById('endDate').value = endDate.toISOString().split('T')[0];
            }
        });

        // Initialize current date
        document.getElementById('startDate').value = new Date().toISOString().split('T')[0];

        console.log('JoFé+ Project Creation Page initialized');
    </script>
