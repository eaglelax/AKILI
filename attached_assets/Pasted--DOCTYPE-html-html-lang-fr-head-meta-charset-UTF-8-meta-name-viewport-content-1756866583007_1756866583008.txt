            if (validateCurrentStep()) {
                if (currentStep < 4) {
                    showStep(currentStep + 1);
                    if (currentStep === 4) {
                        updateSummary();
                    }
                } else {
                    createProject();
                }
            }
        });

        // Step validation
        function validateCurrentStep() {
            switch (currentStep) {
                case 1:
                    const name = document.getElementById('projectName').value;
                    const client = document.getElementById('clientSelect').value;
                    const type = document.getElementById('projectType').value;
                    
                    if (!name || !client || !type) {
                        showNotification('Veuillez remplir tous les champs obligatoires', 'error');
                        return false;
                    }
                    
                    // Update project data
                    projectData.name = name;
                    projectData.client = client;
                    projectData.type = type;
                    projectData.description = document.getElementById('projectDescription').value;
                    projectData.budget = document.getElementById('clientBudget').value || 0;
                    projectData.startDate = document.getElementById('startDate').value;
                    projectData.endDate = document.getElementById('endDate').value;
                    projectData.priority = document.getElementById('priority').value;
                    
                    return true;
                    
                case 2:
                    if (selectedMembers.length === 0) {
                        showNotification('Veuillez sélectionner au moins un membre d\'équipe', 'error');
                        return false;
                    }
                    projectData.team = selectedMembers;
                    return true;
                    
                case 3:
                    if (projectTasks.length === 0) {
                        showNotification('Veuillez ajouter au moins une tâche', 'error');
                        return false;
                    }
                    projectData.tasks = projectTasks;
                    return true;
                    
                case 4:
                    return true;
                    
                default:
                    return true;
            }
        }

        // Update team costs
        function updateTeamCosts() {
            const breakdown = document.getElementById('teamCostBreakdown');
            const hourlyCost = document.getElementById('teamHourlyCost');
            const estimatedCost = document.getElementById('teamEstimatedCost');
            
            if (selectedMembers.length === 0) {
                breakdown.innerHTML = '<p class="text-sm text-gray-500">Équipe non constituée</p>';
                hourlyCost.textContent = '0 FCFA';
                estimatedCost.textContent = '0 FCFA';
                return;
            }
            
            let totalHourly = 0;
            const breakdownHtml = selectedMembers.map(memberId => {
                const member = teamMembers[memberId];
                totalHourly += member.rate;
                return `
                    <div class="flex justify-between text-sm">
                        <span>${member.name}:</span>
                        <span>${member.rate.toLocaleString()} FCFA/h</span>
                    </div>
                `;
            }).join('');
            
            breakdown.innerHTML = breakdownHtml;
            hourlyCost.textContent = totalHourly.toLocaleString() + ' FCFA';
            estimatedCost.textContent = (totalHourly * 40).toLocaleString() + ' FCFA';
        }

                                Coût: <span id="taskCost${task.id}">0 FCFA</span>

                </div>

            document.getElementById('tasksList').insertAdjacentHTML('beforeend', taskHtml);
            projectTasks.push(task);
            updateTaskCost(task.id);
            updateProjectSummary();
        }

        function updateTask(taskId, field, value) {
            const task = projectTasks.find(t => t.id === taskId);
            if (task) {
                task[field] = value;
                updateTaskCost(taskId);
                updateProjectSummary();
            }
        }

        function updateTaskCost(taskId) {
            const task = projectTasks.find(t => t.id === taskId);
            if (task && task.member) {
                const member = teamMembers[task.member];
                const cost = task.duration * member.rate;
                document.getElementById(`taskCost${taskId}`).textContent = cost.toLocaleString() + ' FCFA';
            }
        }

        function removeTask(taskId) {
            projectTasks = projectTasks.filter(t => t.id !== taskId);
            document.querySelector(`[data-task-id="${taskId}"]`).remove();
            updateProjectSummary();
        }

        // Project summary updates
        function updateProjectSummary() {
            const totalTasks = projectTasks.length;
            const totalDuration = projectTasks.reduce((sum, task) => sum + task.duration, 0);
            let totalCost = 0;
            
            projectTasks.forEach(task => {
                if (task.member) {
                    const member = teamMembers[task.member];
                    totalCost += task.duration * member.rate;
                }
            });
            
            document.getElementById('totalTasks').textContent = totalTasks;
            document.getElementById('totalDuration').textContent = totalDuration + ' heures';
            document.getElementById('totalProjectCost').textContent = totalCost.toLocaleString() + ' FCFA';
        }

        // Summary update for final step
        function updateSummary() {
            // Basic info
            document.getElementById('summaryName').textContent = projectData.name;
            document.getElementById('summaryClient').textContent = document.getElementById('clientSelect').options[document.getElementById('clientSelect').selectedIndex].text;
            document.getElementById('summaryType').textContent = document.getElementById('projectType').options[document.getElementById('projectType').selectedIndex].text;
            document.getElementById('summaryStartDate').textContent = projectData.startDate || 'Non définie';
            document.getElementById('summaryEndDate').textContent = projectData.endDate || 'Non définie';
            document.getElementById('summaryDescription').textContent = projectData.description || 'Aucune description';
            
            // Team summary
            const teamContainer = document.getElementById('summaryTeam');
            if (selectedMembers.length === 0) {
                teamContainer.innerHTML = '<p class="text-sm text-gray-500">Aucun membre assigné</p>';
            } else {
                const teamHtml = selectedMembers.map(memberId => {
                    const member = teamMembers[memberId];
                    return `
                        <div class="flex items-center justify-between py-2">
                            <span class="text-sm" style="color: var(--jofe-blue-deep);">${member.name}</span>
                            <span class="text-xs" style="color: var(--jofe-blue-medium);">${member.role}</span>
                        </div>
                    `;
                }).join('');
                teamContainer.innerHTML = teamHtml;
            }
            
            // Financial summary
            const budget = parseInt(projectData.budget) || 0;
            let totalCost = 0;
            
            projectTasks.forEach(task => {
                if (task.member) {
                    const member = teamMembers[task.member];
                    totalCost += task.duration * member.rate;
                }
            });
            
            const margin = budget - totalCost;
            const marginRate = budget > 0 ? ((margin / budget) * 100).toFixed(1) : 0;
            
            document.getElementById('finalBudget').textContent = budget.toLocaleString() + ' FCFA';
            document.getElementById('finalCost').textContent = totalCost.toLocaleString() + ' FCFA';
            document.getElementById('finalMargin').textContent = margin.toLocaleString() + ' FCFA';
            document.getElementById('marginRate').textContent = marginRate + '%';
            
            // Create financial chart
            createFinancialChart(budget, totalCost, margin);
        }

        // Create financial chart

            const ctx = document.getElementById('financialChart').getContext('2d');
            
            new Chart(ctx, {
                type: 'doughnut',
                data: {
                    labels: ['Coût Production', 'Marge Brute'],

                        borderWidth: 0
                    }]
                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    plugins: {
                        legend: {
                            position: 'bottom',
                            labels: {
                                padding: 20,
                                usePointStyle: true
                            }
                        }
                    }
                }
            });
        }

        // Create planning chart
        function createPlanningChart() {
            const ctx = document.getElementById('planningChart').getContext('2d');
            
            const memberData = {};
            projectTasks.forEach(task => {
                if (task.member) {
                    const member = teamMembers[task.member];
                    if (!memberData[member.name]) {
                        memberData[member.name] = 0;
                    }
                    memberData[member.name] += task.duration;
                }
            });
            
            new Chart(ctx, {
                type: 'bar',
                data: {
                    labels: Object.keys(memberData),
                    datasets: [{
                        label: 'Heures assignées',
                        data: Object.values(memberData),
                        backgroundColor: '#37B6E9',
                        borderColor: '#3475BB',
                        borderWidth: 1
                    }]
                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    scales: {
                        y: {
                            beginAtZero: true,
                            ticks: {
                                callback: function(value) {
                                    return value + 'h';
                                }
                            }
                        }
                    },
                    plugins: {
                        legend: {
                            display: false
                        }
                    }
                }
            });
        }

        // Auto-save functionality
        function saveDraft() {
            const draftData = {
                ...projectData,
                selectedMembers,
                projectTasks,
                currentStep
            };
            
            localStorage.setItem('jofe_project_draft', JSON.stringify(draftData));
            showNotification('Brouillon sauvegardé', 'success');
        }

        function loadDraft() {
            const draft = localStorage.getItem('jofe_project_draft');
            if (draft) {
                const draftData = JSON.parse(draft);
                projectData = draftData;
                selectedMembers = draftData.selectedMembers || [];
                projectTasks = draftData.projectTasks || [];
                
                // Restore form values
                Object.keys(projectData).forEach(key => {
                    const element = document.getElementById(key) || document.getElementById(`project${key.charAt(0).toUpperCase() + key.slice(1)}`);
                    if (element && projectData[key]) {
                        element.value = projectData[key];
                    }
                });
                
                showNotification('Brouillon chargé', 'info');
            }
        }
