        function generateTimelineScale(startDate, endDate) {
            const scale = document.getElementById('timelineScale');
            scale.innerHTML = '';
            
            let current = new Date(startDate);
            const pixelsPerDay = getPixelsPerDay();
            
            while (current <= endDate) {
                const unit = document.createElement('div');
                unit.className = 'timeline-unit';
                unit.style.width = `${pixelsPerDay * getDaysInUnit()}px`;
                
                if (currentZoom === 'day') {
                    unit.textContent = current.toLocaleDateString('fr-FR', { weekday: 'short', day: 'numeric' });

                } else if (currentZoom === 'week') {
                    unit.textContent = `Sem. ${getWeekNumber(current)}`;
                    current.setDate(current.getDate() + 7);
                } else {
                    unit.textContent = current.toLocaleDateString('fr-FR', { month: 'long', year: 'numeric' });
                    current.setMonth(current.getMonth() + 1);
                }
                
                scale.appendChild(unit);
            }

            const pixelsPerDay = getPixelsPerDay();
            let position = 0;
            
            while (current <= endDate) {
                const line = document.createElement('div');
                line.className = 'grid-line';
                line.style.left = `${position}px`;
                grid.appendChild(line);
                
                if (currentZoom === 'day') {
                    current.setDate(current.getDate() + 1);
                    position += pixelsPerDay;
                } else if (currentZoom === 'week') {
                    current.setDate(current.getDate() + 7);
                    position += pixelsPerDay * 7;
                } else {
                    const daysInMonth = new Date(current.getFullYear(), current.getMonth() + 1, 0).getDate();
                    current.setMonth(current.getMonth() + 1);
                    position += pixelsPerDay * daysInMonth;
                }
            }
        }

        function positionTodayLine() {
            const todayLine = document.getElementById('todayLine');
            const today = new Date();
            const startDate = new Date(today.getFullYear(), today.getMonth(), 1);
            const daysDiff = Math.floor((today - startDate) / (1000 * 60 * 60 * 24));
            const pixelsPerDay = getPixelsPerDay();
            
            todayLine.style.left = `${daysDiff * pixelsPerDay}px`;
        }

        function renderTasks() {
            const barsContainer = document.getElementById('ganttBars');
            barsContainer.innerHTML = '';
            
            let rowIndex = 0;
            const rowHeight = 40;
            
            Object.values(ganttData.projects).forEach(project => {

        function createTaskBar(task, topPosition) {
            const bar = document.createElement('div');
            bar.className = `gantt-bar ${getProjectColor(task.id)}`;
            bar.dataset.taskId = task.id;
            
            const startDate = new Date(task.startDate);
            const baseDate = new Date(2024, 0, 1); // January 1, 2024
            const daysDiff = Math.floor((startDate - baseDate) / (1000 * 60 * 60 * 24));
            const pixelsPerDay = getPixelsPerDay();
            
            const left = daysDiff * pixelsPerDay;
            const width = task.duration * pixelsPerDay;
            
            bar.style.left = `${left}px`;
            bar.style.width = `${width}px`;
            bar.style.top = `${topPosition}px`;
            
            // Progress bar
            if (task.progress > 0) {
                const progress = document.createElement('div');
                progress.className = 'gantt-progress';
                progress.style.width = `${task.progress}%`;
                bar.appendChild(progress);
            }
            
            // Task name
            const text = document.createElement('span');
            text.textContent = task.name;
            bar.appendChild(text);
            
            // Add drag functionality
            makeDraggable(bar);
            
            return bar;
        }

        function makeDraggable(element) {
            let isDragging = false;
            let startX = 0;
            let startLeft = 0;
            
            element.addEventListener('mousedown', function(e) {
                isDragging = true;
                startX = e.clientX;
                startLeft = parseInt(element.style.left, 10);
                element.classList.add('dragging');
                document.addEventListener('mousemove', handleMouseMove);
                document.addEventListener('mouseup', handleMouseUp);
                e.preventDefault();
            });
            
            function handleMouseMove(e) {
                if (!isDragging) return;
                
                const deltaX = e.clientX - startX;
                const newLeft = Math.max(0, startLeft + deltaX);
                element.style.left = `${newLeft}px`;
                
                // Update task data
                updateTaskDate(element.dataset.taskId, newLeft);
            }
            
            function handleMouseUp() {
                if (isDragging) {
                    isDragging = false;
                    element.classList.remove('dragging');
                    document.removeEventListener('mousemove', handleMouseMove);
                    document.removeEventListener('mouseup', handleMouseUp);
                    
                    // Recalculate dependencies
                    renderDependencies();
                    showNotification('Tâche déplacée avec succès', 'success');
                }
            }
        }

        function updateTaskDate(taskId, leftPosition) {
            const pixelsPerDay = getPixelsPerDay();
            const daysDiff = Math.floor(leftPosition / pixelsPerDay);
            const baseDate = new Date(2024, 0, 1);
            const newDate = new Date(baseDate);
            newDate.setDate(newDate.getDate() + daysDiff);
            
            // Find and update task
            Object.values(ganttData.projects).forEach(project => {
                const task = project.tasks.find(t => t.id === taskId);
                if (task) {
                    task.startDate = newDate.toISOString().split('T')[0];
                }
            });
        }

        function renderDependencies() {
            const container = document.getElementById('dependencies');
            container.innerHTML = '';
            
            Object.values(ganttData.projects).forEach(project => {
                project.tasks.forEach(task => {
                    if (task.dependencies) {
                        task.dependencies.forEach(depId => {
                            const line = createDependencyLine(depId, task.id);
                            if (line) container.appendChild(line);
                        });
                    }
                });
            });
        }

        function createDependencyLine(fromTaskId, toTaskId) {
            const fromBar = document.querySelector(`[data-task-id="${fromTaskId}"]`);
            const toBar = document.querySelector(`[data-task-id="${toTaskId}"]`);
            
            if (!fromBar || !toBar) return null;
            
            const fromRect = fromBar.getBoundingClientRect();
            const toRect = toBar.getBoundingClientRect();
            const containerRect = document.getElementById('timelineBody').getBoundingClientRect();
            
            const line = document.createElement('div');
            line.className = 'dependency-line';
            
            const fromX = fromRect.right - containerRect.left;
            const fromY = fromRect.top - containerRect.top + fromRect.height / 2;
            const toX = toRect.left - containerRect.left;
            const toY = toRect.top - containerRect.top + toRect.height / 2;
            
            // Simple horizontal line for now
            line.style.left = `${fromX}px`;
            line.style.top = `${fromY}px`;
            line.style.width = `${toX - fromX}px`;
            line.style.height = '2px';
            
            return line;
        }

        function getProjectColor(taskId) {
            if (taskId.startsWith('moov')) return 'moov';
            if (taskId.startsWith('bank')) return 'bank';
            if (taskId.startsWith('sunu')) return 'sunu';
            if (taskId.startsWith('roxgold')) return 'roxgold';
            return 'moov';
        }

        function getDaysInUnit() {

        function getWeekNumber(date) {
            const firstDayOfYear = new Date(date.getFullYear(), 0, 1);
            const pastDaysOfYear = (date - firstDayOfYear) / 86400000;
            return Math.ceil((pastDaysOfYear + firstDayOfYear.getDay() + 1) / 7);
        }

        function autoSchedule() {
            showNotification('Planification automatique en cours...', 'info');
            
            setTimeout(() => {
                // Simulate auto-scheduling
                Object.values(ganttData.projects).forEach(project => {
                    project.tasks.forEach((task, index) => {
                        if (index > 0) {
                            const prevTask = project.tasks[index - 1];
                            const prevEndDate = new Date(prevTask.startDate);
                            prevEndDate.setDate(prevEndDate.getDate() + prevTask.duration);
                            task.startDate = prevEndDate.toISOString().split('T')[0];
                        }
                    });
                });
                
                renderGantt();
                showNotification('Planification automatique terminée', 'success');
            }, 2000);
        }

            for (const project of Object.values(ganttData.projects)) {

            return null;
        }

                    const taskId = taskRow.dataset.task;
                    
                    // Highlight corresponding gantt bar
                    document.querySelectorAll('.gantt-bar').forEach(bar => {
                        bar.classList.remove('selected');
                    });
                    
                    const correspondingBar = document.querySelector(`[data-task-id="${taskId}"]`);
                    if (correspondingBar) {
                        correspondingBar.classList.add('selected');
                    }
                    
                    // Highlight task row
                    document.querySelectorAll('.task-row').forEach(row => {
                        row.classList.remove('selected');
                    });
                    taskRow.classList.add('selected');
                }
            });

                    exportGantt();

                    closeTaskModal();
                }
            });
