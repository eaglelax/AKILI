        // Template selection
        let selectedTemplateId = null;
        
        function selectTemplate(templateId) {
            // Remove previous selection
            document.querySelectorAll('.template-card').forEach(card => {
                card.classList.remove('selected');
            });
            
            // Select new template
            event.currentTarget.classList.add('selected');
            selectedTemplateId = templateId;
            
            // Update summary
            const templateNames = {
                'campagne360': 'Campagne 360°',
                'identite': 'Identité Visuelle',
                'digital': 'Digital & Réseaux',
                'custom': 'Projet Personnalisé'
            };
            
            document.getElementById('selectedTemplate').textContent = templateNames[templateId];
            
            // Pre-fill form based on template
            const templateDefaults = {
                'campagne360': {
                    type: 'campagne_360',
                    duration: '6-8 semaines'
                },
                'identite': {
                    type: 'identite_visuelle',
                    duration: '3-4 semaines'
                },
                'digital': {
                    type: 'digital_social',
                    duration: 'En continu'
                },
                'custom': {
                    type: '',
                    duration: 'Variable'
                }
            };
            
            if (templateDefaults[templateId]) {
                document.getElementById('projectType').value = templateDefaults[templateId].type;
                document.getElementById('durationSummary').textContent = templateDefaults[templateId].duration;
            }
        }
        
        // Update budget summary
        document.getElementById('projectBudget').addEventListener('input', function() {
            const budget = parseInt(this.value) || 0;
            document.getElementById('budgetSummary').textContent = budget.toLocaleString() + ' FCFA';
        });
        
        // Update team summary
        document.querySelectorAll('input[type="checkbox"]').forEach(checkbox => {
            checkbox.addEventListener('change', function() {
                const checkedBoxes = document.querySelectorAll('input[type="checkbox"]:checked').length;
                document.getElementById('teamSummary').textContent = checkedBoxes + ' membre(s)';
            });
        });
        
        // Save project
        function saveProject() {
            const formData = new FormData(document.getElementById('projectForm'));

            event.target.innerHTML = '<div class="flex items-center"><div class="animate-spin rounded-full h-4 w-4 border-b-2 border-white"></div><span class="ml-2">Sauvegarde...</span></div>';
            
            setTimeout(() => {
                event.target.innerHTML = '<svg class="w-4 h-4 mr-2" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M5 13l4 4L19 7"></path></svg>Sauvegardé';

                }, 2000);
            }, 1500);
        }
        
        // Create project
        function createProject() {
            const form = document.getElementById('projectForm');
            
            if (!form.checkValidity()) {
                form.reportValidity();
                return;
            }
            
            // Show loading
            event.target.innerHTML = '<div class="flex items-center"><div class="animate-spin rounded-full h-4 w-4 border-b-2 border-white"></div><span class="ml-2">Création...</span></div>';
            
            setTimeout(() => {
                // Show success and redirect
                alert('✅ Projet créé avec succès !');
                // In real app: window.location.href = '/projets';
            }, 2000);
        }

            const nextMonth = new Date();
            nextMonth.setMonth(nextMonth.getMonth() + 1);

        });
