    <script>
        // Timer functionality
        let timerInterval;
        let startTime = Date.now() - (2 * 3600 + 34 * 60 + 12) * 1000; // 2h34m12s ago
        let isRunning = true;

        function updateTimer() {
            if (isRunning) {
                const elapsed = Date.now() - startTime;
                const hours = Math.floor(elapsed / 3600000);
                const minutes = Math.floor((elapsed % 3600000) / 60000);
                const seconds = Math.floor((elapsed % 60000) / 1000);
                
                document.getElementById('currentTimer').textContent = 
                    `${hours.toString().padStart(2, '0')}:${minutes.toString().padStart(2, '0')}:${seconds.toString().padStart(2, '0')}`;
            }
        }

        // Update timer every second
        timerInterval = setInterval(updateTimer, 1000);

        // Timer controls
        document.getElementById('playBtn').addEventListener('click', function() {
            if (!isRunning) {
                startTime = Date.now() - (Date.now() - startTime);
                isRunning = true;
                this.disabled = true;
                document.getElementById('pauseBtn').disabled = false;
            }
        });

        document.getElementById('pauseBtn').addEventListener('click', function() {
            if (isRunning) {
                isRunning = false;
                this.disabled = true;
                document.getElementById('playBtn').disabled = false;
            }
        });

        document.getElementById('stopBtn').addEventListener('click', function() {
            if (confirm('Êtes-vous sûr de vouloir arrêter définitivement ce timer ?')) {
                isRunning = false;
                clearInterval(timerInterval);
                document.getElementById('currentTimer').textContent = '00:00:00';
            }
        });

        // File drag and drop
        const fileDropZone = document.querySelector('.file-attachment');
        
        fileDropZone.addEventListener('dragover', function(e) {
            e.preventDefault();
            this.style.borderColor = 'var(--jofe-blue-light)';
            this.style.backgroundColor = 'rgba(55, 182, 233, 0.05)';
        });

        fileDropZone.addEventListener('dragleave', function(e) {
            e.preventDefault();
            this.style.borderColor = 'var(--jofe-gray)';
            this.style.backgroundColor = 'transparent';
        });

        fileDropZone.addEventListener('drop', function(e) {
            e.preventDefault();
            this.style.borderColor = 'var(--jofe-gray)';
            this.style.backgroundColor = 'transparent';
            
            // Handle file upload here
            console.log('Files dropped:', e.dataTransfer.files);
        });

        // Initialize timer
        updateTimer();
    </script>
